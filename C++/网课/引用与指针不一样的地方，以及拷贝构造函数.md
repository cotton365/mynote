```cpp
#include <iostream>
#include <string>
using namespace std;

class Person {
private:
    string name;
    int age;

public:
    // 1. 普通构造函数
    Person(const string& n, int a) : name(n), age(a) {
        cout << "[构造] " << name << " (this=" << this << ")" << endl;
    }

    // 2. 转换构造函数：int -> Person，用于演示隐式转换
    Person(int a) : name("临时对象"), age(a) {
        cout << "[转换构造] 临时对象 (this=" << this << ")" << endl;
    }

    // 3. 拷贝构造函数：参数是 const Person&，只绑定实参，不产生新转换
    Person(const Person& other) : name(other.name), age(other.age) {
        cout << "[拷贝构造] 实参地址=" << &other
             << " -> 新对象地址=" << this << endl;
    }

    ~Person() {
        cout << "[析构] " << name << " (this=" << this << ")" << endl;
    }
};

int main() {
    cout << "--- 1. 创建 p1 ---" << endl;
    Person p1("张三", 20);
    cout << "p1 地址: " << &p1 << endl;

    cout << "\n--- 2. 拷贝构造 p2(p1) ---" << endl;
    Person p2(p1);  // p1 是 Person，直接绑定到 const Person&，不会调用转换构造
    cout << "p2 地址: " << &p2 << endl;

    cout << "\n--- 3. 隐式转换：const Person& ref = 25 ---" << endl;
    const Person& ref = 25;  // int 先转换成临时 Person，再绑定给引用
    cout << "ref 绑定的临时对象地址: " << &ref << endl;

    cout << "\n--- main 结束 ---" << endl;
    return 0;
}
```
**这里面的拷贝函数使用的是让形参里面的这个值赋值你填入的这个person对象
